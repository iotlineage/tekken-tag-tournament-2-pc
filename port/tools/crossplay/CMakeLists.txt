cmake_minimum_required(VERSION 3.20)
project(ttt2_crossplay_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(TTT2_CROSSPLAY ON CACHE BOOL "")
include(../../cmake/Crossplay.cmake)
if(NOT WIN32)
    find_package(OpenSSL REQUIRED)
endif()
find_package(Threads REQUIRED)
set(kernel ../../src/runtime/kernel)
add_executable(crossplay_probe crossplay_probe.cpp ${kernel}/community_net.cpp ${kernel}/community_tls.cpp ${kernel}/server_list.cpp)
target_include_directories(crossplay_probe PRIVATE ${kernel})
target_link_libraries(crossplay_probe PRIVATE Threads::Threads)
if(NOT WIN32)
    target_link_libraries(crossplay_probe PRIVATE OpenSSL::SSL OpenSSL::Crypto)
endif()
if(WIN32)
    target_link_libraries(crossplay_probe PRIVATE ws2_32 secur32 crypt32 bcrypt)
endif()
ttt2_enable_crossplay(crossplay_probe)
